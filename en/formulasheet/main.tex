\begin{half}{Physics 1}{Formulas}
\topic{Motion and forces}
\begin{formulas}
$v = at + v_0$ & $s = \dfrac{v + v_0}{2}\,t$ & \\
$v^2 - v_0^2 = 2as$ & $s = \frac{1}{2}at^2 + v_0 t$ & \\
\end{formulas}
\formularow{$\Sigma F = 0 \iff v$ constant and rectilinear}
\begin{formulas}
$\Sigma\vv{F} = m\vv{a}$ & $F^{*} = F$ & $G = mg$ \\
$R = \mu N$ & $B = \rho_f V g$ & $L = kv^2$ \\
$\vv{p} = m\vv{v}$ & $\Sigma\vv{F}\cdot t = \Delta\vv{p}$ & \\
\end{formulas}
\formularow{$m_A v_A + m_B v_B = m_A v_{A0} + m_B v_{B0}$}

\topic{Mechanical energy and power}
\begin{formulas}
$W = Fs\cos\theta$ & $W_{\Sigma F} = \Delta E_k$ & $P = W/t$ \\
$E_k = \frac{1}{2}mv^2$ & $E_p = mgh$ & $E = E_k + E_p$ \\
\end{formulas}
\formularow{$\frac{1}{2}mv^2 + mgh = \frac{1}{2}mv_0^2 + mgh_0$}
\formularow{$\eta = \dfrac{\text{useful energy}}{\text{supplied energy}} = \dfrac{\text{useful power}}{\text{supplied power}}$}

\topic{Pressure and temperature}
\begin{formulas}
$p = F/A$ & $T = t + 273$ & $E_k = \frac{3}{2}kT$ \\
$c = \dfrac{Q}{m\,\Delta T}$ & $C = \dfrac{Q}{\Delta T}$ & $\Delta U = W + Q$ \\
\end{formulas}
\columnbreak
\topic{Pressure and temperature (cont.)}
\begin{formulastwo}
$\mathit{COP} = \dfrac{Q_{\text{out}}}{E_{\text{electric}}}$ &
$\mathit{COP}_{\text{Carnot}} = \dfrac{T_{\text{out}}}{T_{\text{out}} - T_{\text{in}}}$ \\
\end{formulastwo}

\topic{Waves and radiation}
\begin{formulas}
$f = 1/T$ & $v = \lambda f$ & $U = P/A$ \\
$\lambda_{\text{peak}}\cdot T = a$ & $U = \sigma T^4$ & $I = \dfrac{P}{4\pi r^2}$ \\
\end{formulas}

\topic{Atomic and nuclear physics}
\begin{formulas}
$E = hf = \dfrac{hc}{\lambda}$ & $E_n = -\dfrac{B}{n^2}$ & \\
\end{formulas}
\begin{formulastwo}
$hf = E_n - E_m,\quad n > m$ & $hf = B\left(\dfrac{1}{m^2} - \dfrac{1}{n^2}\right)$ \\
\end{formulastwo}
\begin{formulas}
$A = Z + N$ & $E_0 = mc^2$ & $\Delta E_0 = \Delta m\cdot c^2$ \\
\end{formulas}

\topic{Electricity}
\begin{formulas}
$I = \dfrac{Q}{t}$ & $U = \dfrac{W}{Q}$ & $R = \dfrac{U}{I}$ \\
$W = UIt$ & $P = UI = RI^2 = \dfrac{U^2}{R}$ & \\
\end{formulas}
\formularow{Series: \quad $R_{\text{tot}} = R_1 + R_2 + \cdots + R_n$}
\formularow{Parallel: \quad $R_{\text{tot}} = \left(\dfrac{1}{R_1} + \dfrac{1}{R_2} + \cdots + \dfrac{1}{R_n}\right)^{-1}$}
\end{half}
